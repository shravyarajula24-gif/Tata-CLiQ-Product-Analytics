def z_stats(sample_mean,pop_std,sample_size,confidence_level):

    area = (1 + confidence_level) / 2
    print(f'Area is {area}')

    z_param = st.norm.ppf(area)
    #ppf = percent point function
    #it is used to calculate the a value for the given area
    print(f'Z parameter from the table is {z_param}')

    CI = [sample_mean - z_param*(pop_std/(sample_size**0.5))],[sample_mean + (pop_std/(sample_size**0.5))]
    print(f'Confidence interval is {CI}')

    MOE = z_param*(pop_std/(sample_size**0.5))
    print('moe',MOE)

def t_stats(sample_mean,sample_std,sample_size,confidence_level):

    area = (1 + confidence_level) / 2
    print(f'Area is {area}')

    t_param = st.t.ppf(area,df=(sample_size - 1))
    print(f't parameter is {t_param}')

    CI = [sample_mean - t_param*(sample_std/(sample_size**0.5))],[sample_mean + (sample_std/(sample_size**0.5))]
    print(f'Confidence interval is {CI}')

    MOE = t_param*(sample_std/(sample_size**0.5))
    print('moe',MOE)

sample_df = df.sample(100, random_state=42)
sample_mean = sample_df['Price'].mean()
sample_size = len(sample_df)
pop_std = df['Price'].std(ddof=0)
sample_std = sample_df['Price'].std()

# Z-test
z_crit = st.norm.ppf(1 - alpha/2)
moe_z = z_crit * pop_std / np.sqrt(sample_size)
z_stat = (sample_mean - hypothesized_mean) / (pop_std / np.sqrt(sample_size))
p_value_z = 2 * st.norm.sf(abs(z_stat))
print("Z-statistic:", z_stat)
print("P-value:", p_value_z)
print("95% CI:", (sample_mean - moe_z, sample_mean + moe_z))
print("Reject H0" if p_value_z < alpha else "Fail to reject H0")

# T-test
t_crit = st.t.ppf(1 - alpha/2, sample_size - 1)
moe_t = t_crit * sample_std / np.sqrt(sample_size)

p_value_t = 2 * st.t.sf(abs(t_stat), sample_size - 1)

print("P-value:", p_value_t)
print("95% CI:", (sample_mean - moe_t, sample_mean + moe_t))
print("Reject H0" if p_value_t < alpha else "Fail to reject H0")
